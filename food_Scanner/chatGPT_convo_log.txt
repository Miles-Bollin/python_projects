import cv2


def main():
    image = cv2.imread("barcode_test.jpg")

    if image is None:
        print("Error: Could not load barcode_test.jpg")
        return

    detector = cv2.barcode.BarcodeDetector()

    detected, points = detector.detect(image)

    if not detected:
        print("No barcode detected.")
        return

    print("Barcode location detected.")

    points = points[0]

    x_coordinates = points[:, 0]
    y_coordinates = points[:, 1]

    x_min = int(x_coordinates.min())
    x_max = int(x_coordinates.max())
    y_min = int(y_coordinates.min())
    y_max = int(y_coordinates.max())

    padding = 20

    x_min = max(0, x_min - padding)
    y_min = max(0, y_min - padding)
    x_max = min(image.shape[1], x_max + padding)
    y_max = min(image.shape[0], y_max + padding)

    barcode = image[y_min:y_max, x_min:x_max]

    cv2.imwrite("detected_barcode.jpg", barcode)

    print(f"Barcode crop saved.")
    print(f"Crop dimensions: {barcode.shape}")

    decoded_info, points, straight_code = (
        detector.detectAndDecode(barcode)
    )

    if decoded_info:
        print(f"Barcode decoded: {decoded_info}")
    else:
        print("Barcode was detected but could not be decoded.")


if __name__ == "__main__":
    main()
